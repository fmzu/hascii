```ts
export const parseDotStringToColorGrid = (
  dots: string,
  palette: string[],
): (string | null)[][] => {
  if (dots === "") {
    return []
  }

  const tokens = dots.split("-")
  const size = Math.floor(Math.sqrt(tokens.length))

  if (size === 0) {
    return []
  }

  const grid: (string | null)[][] = []

  for (let y = 0; y < size; y++) {
    const row: (string | null)[] = []
    for (let x = 0; x < size; x++) {
      const index = Number.parseInt(tokens[y * size + x], 10)
      if (Number.isNaN(index) || index < 0 || index >= palette.length) {
        row.push(null)
      } else {
        row.push(palette[index])
      }
    }
    grid.push(row)
  }

  return grid
}
```

const OGP_WIDTH = 1200
const OGP_HEIGHT = 630

export const generateOgpSvg = (
  colorGrid: (string | null)[][],
): string => {
  const rows = colorGrid.length
  const cols = rows === 0 ? 0 : colorGrid[0].length
  const gridSize = Math.max(rows, cols)

  const scale = gridSize === 0 ? 1 : Math.floor(OGP_HEIGHT / gridSize)
  const artWidth = cols * scale
  const artHeight = rows * scale
  const offsetX = Math.floor((OGP_WIDTH - artWidth) / 2)
  const offsetY = Math.floor((OGP_HEIGHT - artHeight) / 2)

  const rects: string[] = []

  for (let y = 0; y < rows; y++) {
    for (let x = 0; x < cols; x++) {
      const color = colorGrid[y][x]
      if (color === null) {
        continue
      }
      rects.push(
        `<rect x="${offsetX + x * scale}" y="${offsetY + y * scale}" width="${scale}" height="${scale}" fill="${color}"/>`,
      )
    }
  }

  return [
    `<svg xmlns="http://www.w3.org/2000/svg" width="${OGP_WIDTH}" height="${OGP_HEIGHT}">`,
    `<rect width="${OGP_WIDTH}" height="${OGP_HEIGHT}" fill="#ffffff"/>`,
    ...rects,
    "</svg>",
  ].join("")
}
```

import { Resvg, initWasm } from "@resvg/resvg-wasm"
// @ts-ignore — resvg-wasm の index_bg.wasm を直接 import
import resvgWasm from "@resvg/resvg-wasm/index_bg.wasm"

let initialized = false

export const renderSvgToPng = async (svg: string): Promise<Uint8Array> => {
  if (!initialized) {
    await initWasm(resvgWasm)
    initialized = true
  }

  const resvg = new Resvg(svg, {
    fitTo: { mode: "width", value: 1200 },
  })

  const rendered = resvg.render()
  return rendered.asPng()
}
```

```ts
import { vValidator } from "@hono/valibot-validator"
import { object, string } from "valibot"
import { apiFactory } from "~/interface/api-factory"
import { generateOgpSvg } from "~/lib/generate-ogp-svg"
import { parseDotStringToColorGrid } from "~/lib/parse-dots-to-color-grid"
import { renderSvgToPng } from "~/lib/render-svg-to-png"
import { xtermColors } from "~/lib/xterm-colors"

export const GET = apiFactory.createHandlers(
  vValidator("param", object({ dots: string() })),
  async (c) => {
    const dots = c.req.param("dots")

    const cacheKey = new Request(c.req.url)
    const cache = caches.default

    const cached = await cache.match(cacheKey)
    if (cached) {
      return cached
    }

    const colorGrid = parseDotStringToColorGrid(dots, xtermColors)
    const svg = generateOgpSvg(colorGrid)
    const png = await renderSvgToPng(svg)

    const response = new Response(png, {
      headers: {
        "Content-Type": "image/png",
        "Cache-Control": "public, max-age=31536000, immutable",
      },
    })

    c.executionCtx.waitUntil(cache.put(cacheKey, response.clone()))

    return response
  },
)
```
